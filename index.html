    window.addEventListener('DOMContentLoaded', () => {

        class Particle {

            constructor() {

                this.x = Math.random() * window.innerWidth;

                this.y = Math.random() * window.innerHeight;

                this.destX = this.x;

                this.destY = this.y;

                this.vx = 0; this.vy = 0;

                this.acc = 0.004;

                this.friction = 0.92;

                this.size = Math.random() * 1.5 + 0.5;

                this.color = '#fff';

                this.alpha = Math.random();

                this.isHeart = false;

                this.heartIdx = 0;

            }

            update() {

                if (this.isHeart && stage === 3) {

                    const angle = (this.heartIdx / 400) * Math.PI * 2 + heartAngleOffset;

                    const scale = Math.min(width/40, 12);

                    this.destX = width / 2 + (16 * Math.pow(Math.sin(angle), 3)) * scale;

                    this.destY = height / 2 - (13 * Math.cos(angle) - 5 * Math.cos(2 * angle) - 2 * Math.cos(3 * angle) - Math.cos(4 * angle)) * scale;

                }


                let dx = this.destX - this.x;

                let dy = this.destY - this.y;

                this.vx += dx * this.acc;

                this.vy += dy * this.acc;

                this.vx *= this.friction;

                this.vy *= this.friction;

                this.x += this.vx;

                this.y += this.vy;


                this.alpha = 0.4 + Math.abs(Math.sin(Date.now() * 0.001 + this.x * 0.01)) * 0.5;

            }

            draw() {

                ctx.globalAlpha = this.alpha;

                ctx.fillStyle = this.color;

                ctx.beginPath();

                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);

                ctx.fill();

            }

        }


        class Meteor {

            constructor() { this.init(); }

            init() {

                this.x = Math.random() * width;

                this.y = -20;

                this.speed = Math.random() * 4 + 4;

                this.len = Math.random() * 60 + 20;

                this.angle = Math.PI / 4;

            }

            update() {

                this.x += Math.cos(this.angle) * this.speed;

                this.y += Math.sin(this.angle) * this.speed;

                if (this.x > width || this.y > height) this.init();

            }

            draw() {

                ctx.strokeStyle = "rgba(255, 255, 255, 0.2)";

                ctx.lineWidth = 1;

                ctx.beginPath();

                ctx.moveTo(this.x, this.y);

                ctx.lineTo(this.x - Math.cos(this.angle) * this.len, this.y - Math.sin(this.angle) * this.len);

                ctx.stroke();

            }

        }


        function getHandDrawnPoints() {

            const pts = [];

            const cx = width / 2, cy = height / 2 + 50;

            const s = Math.min(width, height) / 450;

            function addLine(x1, y1, x2, y2, color, density = 0.02) {

                for(let i=0; i<1; i+=density) {

                    pts.push({

                        x: cx + (x1 + (x2 - x1) * i)*s + (Math.random()-0.5)*5,

                        y: cy + (y1 + (y2 - y1) * i)*s + (Math.random()-0.5)*5,

                        c: color

                    });

                }

            }

            addLine(-50, -120, -20, -135, '#ffe16b'); // 头发

            addLine(-20, -135, 10, -120, '#ffe16b');

            addLine(-55, -110, -45, -60, '#ffcc33'); // 披风

            addLine(-90, -40, -100, 0, '#ffcc33');

            addLine(-45, -60, -40, 60, '#fff7d1'); // 身体

            addLine(35, -20, 50, -50, '#ff8c00'); // 狐狸

            addLine(50, -50, 65, -20, '#ff8c00');

            addLine(30, 60, 70, 60, '#ff8c00');

            addLine(70, 60, 130, 60, '#ffb366');

            return pts;

        }


        function getTextPoints(text, size) {

            const pts = [];

            const tCanvas = document.createElement('canvas');

            const tCtx = tCanvas.getContext('2d');

            tCanvas.width = width; tCanvas.height = height;

            tCtx.fillStyle = "white";

            tCtx.textAlign = "center";

            tCtx.textBaseline = "middle";

            tCtx.font = `bold ${size}px sans-serif`;

            tCtx.fillText(text, width/2, height/2);

            const data = tCtx.getImageData(0, 0, width, height).data;

            for (let y = 0; y < height; y += 5) {

                for (let x = 0; x < width; x += 5) {

                    if (data[(y * width + x) * 4 + 3] > 128) pts.push({x, y});

                }

            }

            return pts;

        }


        async function updateStage(s) {

            stage = s;

            let targets = [];

            let color = "#fff";


            if (s === -2) {

                targets = getHandDrawnPoints();

            } else if (s === 3) {

                targets = getTextPoints("Happy Birthday", Math.min(width/10, 70));

                color = "#ffbaba";

            } else if (s === 5) {

                const cx = width/2, cy = height/2 + 100;

                for(let x=-120; x<120; x+=7) for(let y=0; y<40; y+=7) targets.push({x: cx+x, y: cy+y, c: '#ff99cc'});

                for(let x=-80; x<80; x+=7) for(let y=-40; y<0; y+=7) targets.push({x: cx+x, y: cy+y, c: '#ffcce6'});

                for(let x=-50; x<50; x+=7) for(let y=-80; y<-40; y+=7) targets.push({x: cx+x, y: cy+y, c: '#ffffff'});

                const cXs = [-35, 0, 35];

                cXs.forEach(x => {

                    for(let y=-110; y<-80; y+=5) targets.push({x: cx+x, y: cy+y, c: '#ffd700'});

                });

            }


            particles.forEach((p, i) => {

                p.isHeart = false;

                p.acc = 0.004;

                if (i < targets.length) {

                    p.destX = targets[i].x;

                    p.destY = targets[i].y;

                    p.color = targets[i].c || color;

                } else if (s === 3 && i < targets.length + 400) {

                    p.isHeart = true;

                    p.heartIdx = i - targets.length;

                    p.color = "rgba(255, 80, 120, 0.8)";

                } else {

                    const angle = Math.random() * Math.PI * 2;

                    const r = Math.random() * width * 1.2;

                    p.destX = width/2 + Math.cos(angle) * r;

                    p.destY = height/2 + Math.sin(angle) * r;

                    p.color = "rgba(255,255,255,0.05)";

                }

            });

        }

        class Firework {

            constructor(x, y) {

                this.x = x; this.y = y;

                this.parts = [];

                const hue = Math.random() * 360;

                for(let i=0; i<60; i++) {

                    const a = Math.random() * Math.PI * 2;

                    const s = Math.random() * 4 + 1;

                    this.parts.push({

                        x:x, y:y, vx:Math.cos(a)*s, vy:Math.sin(a)*s,

                        life: 1.0, decay: Math.random()*0.02 + 0.015,

                        c: `hsla(${hue}, 80%, 60%, 1)`

                    });

                }

            }

            update() {

                this.parts.forEach(p => { p.x += p.vx; p.y += p.vy; p.vy += 0.04; p.life -= p.decay; });

                this.parts = this.parts.filter(p => p.life > 0);

            }

            draw() {

                this.parts.forEach(p => {

                    ctx.globalAlpha = p.life; ctx.fillStyle = p.c;

                    ctx.beginPath(); ctx.arc(p.x, p.y, 1.2, 0, Math.PI*2); ctx.fill();

                });

            }

        }


        function init() {

            width = canvas.width = window.innerWidth;

            height = canvas.height = window.innerHeight;

            stars = Array.from({length:400}, () => ({

                x:Math.random()*width, 

                y:Math.random()*height, 

                s:Math.random()*1.2, 

                o:Math.random()

            }));

            particles = Array.from({length:PARTICLE_COUNT}, () => new Particle());

            meteors = Array.from({length:2}, () => new Meteor());

        }


        function animate() {

            ctx.fillStyle = 'rgba(1, 1, 5, 0.2)';

            ctx.fillRect(0, 0, width, height);


            stars.forEach(s => {

                s.o = 0.1 + Math.abs(Math.sin(Date.now()*0.0005 + s.x)) * 0.8;

                ctx.globalAlpha = s.o; ctx.fillStyle = "#fff";

                ctx.beginPath(); ctx.arc(s.x, s.y, s.s, 0, Math.PI*2); ctx.fill();

            });

            ctx.globalAlpha = 1;


            meteors.forEach(m => { m.update(); m.draw(); });


            if (stage === 3) heartAngleOffset += 0.008;

            if (stage === 4) {

                fireworks.forEach((f, i) => { f.update(); f.draw(); if(!f.parts.length) fireworks.splice(i,1); });

            }


            particles.forEach(p => { p.update(); p.draw(); });


            if (stage === 5) {

                const cXs = [width/2-35, width/2, width/2+35];

                cXs.forEach(fx => {

                    const fy = height/2 - 10;

                    const t = Date.now() * 0.006 + fx;

                    ctx.shadowBlur = 15; ctx.shadowColor = "#ff6600";

                    ctx.fillStyle = "#ffcc33";

                    ctx.beginPath();

                    ctx.ellipse(fx, fy + Math.sin(t)*3, 6, 11 + Math.sin(t)*2, 0, 0, Math.PI*2);

                    ctx.fill();

                    ctx.shadowBlur = 0;

                });

            }


            requestAnimationFrame(animate);

        }
